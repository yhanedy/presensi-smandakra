// Logika Login Siswa async function loginSiswa(event) { if (event) event.preventDefault(); // Mencegah reload halaman ke /? const nisnInput = document.getElementById('siswa-nisn') || document.querySelector('input[placeholder*="NISN"]'); const passInput = document.getElementById('siswa-password') || document.querySelector('#form-siswa input[type="password"]'); if (!nisnInput || !passInput) return alert('Input NISN / Password tidak ditemukan!'); const nisn = nisnInput.value.trim(); const password = passInput.value.trim(); if (!nisn || !password) return alert('Silakan isi NISN dan Password!'); try { const response = await fetch(`${API_URL}/api/login`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: nisn, password: password }) }); const result = await response.json(); if (result.success && result.type === 'siswa') { localStorage.setItem('smada_user', JSON.stringify(result.user)); alert(`Selamat datang, ${result.user.nama}!`); // Pindah ke halaman siswa window.location.href = 'siswa.html'; } else { alert(result.message || 'NISN atau Password Salah!'); } } catch (err) { console.error('Error Login:', err); alert('Gagal terhubung ke server.'); } }
